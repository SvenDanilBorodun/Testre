# EduBotics GUI runtime dependencies
#
# pywebview embeds the React UI in a native Edge WebView2 window.
# pythonnet is required by pywebview's `edgechromium` backend on Windows.
#
# Pins below are the lowest versions we know to support Python 3.14.
# pythonnet 3.1.0rc0 is a pre-release (only one that supports 3.14 today),
# so `pip install -r requirements.txt --pre` is required.
pywebview>=6.2.1
pythonnet>=3.1.0rc0

# Native camera bridge: capture the USB cameras directly on Windows (OpenCV
# MSMF, cv2.CAP_MSMF — DSHOW is the EDUBOTICS_CAMERA_BACKEND=dshow rollback,
# see win_camera._capture_backend) and stream JPEG frames into the container.
# The full opencv-python wheel is required (NOT -headless — we need the
# Windows capture backends); it bundles numpy.
opencv-python>=4.8
numpy>=1.24
